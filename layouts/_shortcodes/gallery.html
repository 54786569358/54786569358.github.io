{{- /* Usage inside a leaf bundle (content/gallery/index.md): {{< gallery >}} — renders every image resource of the page Optional: {{< gallery match="*.jpg" >}} */ -}} {{- $match := .Get "match" | default "*" -}} {{- $imgs := .Page.Resources.Match $match -}} {{- $imgs = where $imgs "MediaType.MainType" "image" -}}
{{- range $imgs }} {{- $thumb := . -}} {{- if eq .MediaType.SubType "jpeg" }}{{ $thumb = .Resize "600x q80" }}{{ else if eq .MediaType.SubType "png" }}{{ $thumb = .Resize "600x q80" }}{{ end -}} {{- end }}
